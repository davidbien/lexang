# CMakeLists.txt for the lexang unit tests.
#
# lexangdfatest: Google test based unit tests of the creation of DFAs from regular expressions - see lexangdfatest.cpp.
# lexangtestgen: Generates a lexical analyzer for each of a set of test grammars - see lexangtestgen.cpp.
# lexanglextest: Google test based unit tests which use the generated lexical analyzers - see lexanglextest.cpp.

include(GoogleTest)

if(NOT TARGET gtest)
  include(FetchContent)
  set(GTEST_VERSION 1.14.0 CACHE STRING "Google test version")
  FetchContent_Declare(googletest
    GIT_REPOSITORY https://github.com/google/googletest.git
    GIT_TAG v${GTEST_VERSION})
  # Use the same runtime library settings as we do under MSVC.
  set(gtest_force_shared_crt OFF CACHE BOOL "" FORCE)
  FetchContent_MakeAvailable(googletest)
endif()

add_executable (lexangdfatest lexangdfatest.cpp lexangtest_gen.h lexangtest_types.h lexangtest_tag.h )
target_link_libraries(lexangdfatest lexang gtest gtest_main)
target_compile_definitions( lexangdfatest
  PUBLIC __NDEBUG_THROW
  PUBLIC TRACESENABLED=0
)
gtest_discover_tests(lexangdfatest)

add_executable (lexangtestgen lexangtestgen.cpp lexangtest_gen.h lexangtest_types.h lexangtest_tag.h )
target_link_libraries(lexangtestgen lexang)
target_compile_definitions( lexangtestgen
  PUBLIC __NDEBUG_THROW
  PUBLIC TRACESENABLED=0
)

# The grammars generated by lexangtestgen: each is generated into _lexangtest_<grammar>.h.
# This list must match the grammars generated in lexangtestgen.cpp.
set(LEXANGTEST_GRAMMARS basic triggers triggerstagged triggershistory excludes completes lookahead lookaheadambig lookaheadtwo lookaheadoverlap lookaheadsplit lookaheadvar lookaheadvartagged lookaheadrandom lookaheadprecfwd lookaheadprecrev reject multi tagregisters taghistory tagmixed xmlish xmlishtagged)
set(LEXANGTEST_HEADERS "")
foreach(LEXANGTEST_GRAMMAR ${LEXANGTEST_GRAMMARS})
  list(APPEND LEXANGTEST_HEADERS "${CMAKE_CURRENT_BINARY_DIR}/_lexangtest_${LEXANGTEST_GRAMMAR}.h")
endforeach()
add_custom_command(
  OUTPUT ${LEXANGTEST_HEADERS}
  COMMAND lexangtestgen
  DEPENDS lexangtestgen
  WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}"
  COMMENT "Generating the lexang test lexical analyzers."
)

add_executable (lexanglextest lexanglextest.cpp lexangtest_types.h lexangtest_gen.h lexangtest_tag.h ${LEXANGTEST_HEADERS} )
target_include_directories(lexanglextest PUBLIC ${CMAKE_CURRENT_BINARY_DIR})
target_link_libraries(lexanglextest lexang gtest gtest_main)
target_compile_definitions( lexanglextest
  PUBLIC __NDEBUG_THROW
  PUBLIC TRACESENABLED=0
)
gtest_discover_tests(lexanglextest)
